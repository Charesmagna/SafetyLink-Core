        if (typeof window !== 'undefined') {
            try {
                // we assume dynamic import to avoid blocking and only sync if firebase is ready
                import('../lib/firebase').then(({ db }) => {
                    import('firebase/firestore').then(({ doc, setDoc }) => {
                         const docRef = doc(db, 'incidents', item.id);
                         setDoc(docRef, {
                            id: item.id,
                            latitude: item.lat,
                            longitude: item.lng,
                            description: item.description,
                            org_id: orgId,
                            triggered_by: who,
                            status: 'DISPATCHED',
                            severity: 'CRITICAL',
                            timestamp: new Date(item.timestamp),
                            syncedAt: new Date()
                         }, { merge: true }).catch(e => console.warn('Firebase sync offline failed', e));
                    });
                }).catch(e => console.warn('Firebase module not loaded', e));
            } catch (e) {}
